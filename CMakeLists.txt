cmake_minimum_required(VERSION 3.16)
project(nornHTTP VERSION 0.1.0 LANGUAGES C)
add_library(nornHTTP STATIC src/norn_http.c src/norn_http_request.c src/norn_http_parser.c src/norn_http_stream.c)
target_include_directories(nornHTTP PUBLIC include)
target_compile_features(nornHTTP PUBLIC c_std_99)
option(NORN_HTTP_TESTS "Build host tests" OFF)
option(NORN_HTTP_SANITIZE "Enable ASan and UBSan" OFF)
if(NORN_HTTP_SANITIZE)
  target_compile_options(nornHTTP PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(nornHTTP PUBLIC -fsanitize=address,undefined)
endif()
if(NORN_HTTP_TESTS)
  enable_testing()
  add_executable(test_http tests/test_http.c)
  target_link_libraries(test_http PRIVATE nornHTTP)
  add_test(NAME http COMMAND test_http)
endif()
